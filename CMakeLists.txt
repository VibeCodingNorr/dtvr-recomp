cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DTVR_VERSION LIMIT_COUNT 1)
project(dtvr_recomb VERSION ${DTVR_VERSION} LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.10)

add_library(dtvr-formats STATIC src/sbs.cpp src/layout.cpp src/panel.cpp src/ne_strings.cpp src/worldmap.cpp src/board.cpp src/rules.cpp src/cards.cpp src/economy.cpp src/drago.cpp src/events.cpp src/obstacles.cpp src/ai.cpp src/ai_cards.cpp src/game.cpp src/game_ai.cpp src/game_cards.cpp src/game_drago.cpp src/game_events.cpp src/game_risiko.cpp src/save.cpp)
target_include_directories(dtvr-formats PUBLIC src)
target_compile_features(dtvr-formats PUBLIC cxx_std_17)
target_compile_options(dtvr-formats PRIVATE -Wall -Wextra -Wpedantic)

add_executable(dtvr-recomb src/main.cpp src/play_view.cpp src/sdl_assets.cpp src/game_screen.cpp src/slot_view.cpp src/message_view.cpp src/menu_view.cpp src/card_view.cpp src/text.cpp)
target_compile_definitions(dtvr-recomb PRIVATE DTVR_VERSION="${PROJECT_VERSION}")
target_compile_options(dtvr-recomb PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(dtvr-recomb PRIVATE dtvr-formats PkgConfig::SDL2)
# Optional: text in the game window (Arial or Liberation Sans through SDL2_ttf).
pkg_check_modules(SDL2_TTF QUIET IMPORTED_TARGET SDL2_ttf)
if(SDL2_TTF_FOUND)
  target_compile_definitions(dtvr-recomb PRIVATE DTVR_HAVE_TTF)
  target_link_libraries(dtvr-recomb PRIVATE PkgConfig::SDL2_TTF)
else()
  message(STATUS "SDL2_ttf not found: the game window is built without text")
endif()
