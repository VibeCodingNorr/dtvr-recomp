cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DTVR_VERSION LIMIT_COUNT 1)
project(dtvr_recomb VERSION ${DTVR_VERSION} LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.10)

add_library(dtvr-formats STATIC src/sbs.cpp src/worldmap.cpp src/board.cpp src/rules.cpp src/cards.cpp src/economy.cpp src/drago.cpp src/events.cpp src/obstacles.cpp src/game.cpp src/game_cards.cpp src/game_drago.cpp src/game_events.cpp)
target_include_directories(dtvr-formats PUBLIC src)
target_compile_features(dtvr-formats PUBLIC cxx_std_17)
target_compile_options(dtvr-formats PRIVATE -Wall -Wextra -Wpedantic)

add_executable(dtvr-recomb src/main.cpp src/play_view.cpp)
target_compile_definitions(dtvr-recomb PRIVATE DTVR_VERSION="${PROJECT_VERSION}")
target_compile_options(dtvr-recomb PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(dtvr-recomb PRIVATE dtvr-formats PkgConfig::SDL2)

enable_testing()

target_link_libraries(rules-test PRIVATE dtvr-formats)
add_test(NAME rules COMMAND rules-test)
add_executable(game-test tests/game_test.cpp)
target_compile_options(game-test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(game-test PRIVATE dtvr-formats)
add_test(NAME game COMMAND game-test)
add_executable(cards-test tests/cards_test.cpp)
target_compile_options(cards-test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(cards-test PRIVATE dtvr-formats)

if(Python3_Interpreter_FOUND)
    add_test(NAME iso9660 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_iso9660.py)
