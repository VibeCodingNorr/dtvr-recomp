cmake_minimum_required(VERSION 3.16)
project(dtvr_recomb VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.10)

add_library(dtvr-formats STATIC src/sbs.cpp src/worldmap.cpp src/board.cpp src/rules.cpp)
target_include_directories(dtvr-formats PUBLIC src)
target_compile_features(dtvr-formats PUBLIC cxx_std_17)
target_compile_options(dtvr-formats PRIVATE -Wall -Wextra -Wpedantic)

add_executable(dtvr-recomb src/main.cpp)
target_compile_options(dtvr-recomb PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(dtvr-recomb PRIVATE dtvr-formats PkgConfig::SDL2)

enable_testing()
add_executable(sbs-test tests/sbs_test.cpp)
target_compile_options(sbs-test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sbs-test PRIVATE dtvr-formats)
add_test(NAME sbs COMMAND sbs-test)
add_executable(worldmap-test tests/worldmap_test.cpp)
target_compile_options(worldmap-test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(worldmap-test PRIVATE dtvr-formats)
add_test(NAME worldmap COMMAND worldmap-test)
add_executable(board-test tests/board_test.cpp)
target_compile_options(board-test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(board-test PRIVATE dtvr-formats)
add_test(NAME board COMMAND board-test)
add_executable(rules-test tests/rules_test.cpp)
target_compile_options(rules-test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rules-test PRIVATE dtvr-formats)
add_test(NAME rules COMMAND rules-test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME iso9660 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_iso9660.py)
    add_test(NAME cd-audio COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cd_audio.py)
    add_test(NAME installer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_installer.py)
    add_test(NAME repository-policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_repository_policy.py)
endif()
install(TARGETS dtvr-recomb RUNTIME DESTINATION bin)
install(FILES LICENSE DESTINATION share/licenses/dtvr-recomb)
