cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DTVR_VERSION LIMIT_COUNT 1)
project(dtvr_recomb VERSION ${DTVR_VERSION} LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.10)
pkg_check_modules(VORBISFILE REQUIRED IMPORTED_TARGET vorbisfile)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c>=0.15)
find_package(Threads REQUIRED)

add_library(dtvr-formats STATIC src/sbs.cpp src/layout.cpp src/panel.cpp src/estates.cpp src/save_slots.cpp src/settings.cpp src/ne_strings.cpp src/worldmap.cpp src/board.cpp src/rules.cpp src/cards.cpp src/economy.cpp src/drago.cpp src/events.cpp src/obstacles.cpp src/ai.cpp src/ai_cards.cpp src/game.cpp src/game_ai.cpp src/game_cards.cpp src/game_drago.cpp src/game_events.cpp src/game_risiko.cpp src/save.cpp src/flc.cpp src/vehicle_air.cpp src/vehicle_anim.cpp src/voice_sync.cpp src/intro_script.cpp src/ambient.cpp src/recomp_menu.cpp)
target_include_directories(dtvr-formats PUBLIC src)
target_compile_features(dtvr-formats PUBLIC cxx_std_17)
target_compile_options(dtvr-formats PRIVATE -Wall -Wextra -Wpedantic)

add_executable(dtvr-recomb src/main.cpp src/play_view.cpp src/sdl_assets.cpp src/game_screen.cpp src/game_window.cpp src/display_control.cpp src/recomp_menu_view.cpp src/slot_view.cpp src/message_view.cpp src/menu_view.cpp src/card_view.cpp src/estates_view.cpp src/save_view.cpp src/setup_view.cpp src/stages_view.cpp src/overview_view.cpp src/options_view.cpp src/sound.cpp src/scene.cpp src/finish_view.cpp src/victory_view.cpp src/balance_view.cpp src/stage_view.cpp src/card_shop_view.cpp src/select_player_view.cpp src/flight_view.cpp src/nails_view.cpp src/invest_view.cpp src/risiko_view.cpp src/sb_button.cpp src/window_border.cpp src/map_scroll.cpp src/event_scenes.cpp src/mouse_control_view.cpp src/controller.cpp src/film_view.cpp src/sight_view.cpp src/subtitle_view.cpp src/intro_view.cpp src/text.cpp)
target_compile_definitions(dtvr-recomb PRIVATE DTVR_VERSION="${PROJECT_VERSION}")
target_compile_options(dtvr-recomb PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(dtvr-recomb PRIVATE dtvr-formats PkgConfig::SDL2)
add_library(dtvr-music STATIC src/music_catalog.cpp src/music_stream.cpp src/music_player.cpp src/music_menu.cpp)
target_include_directories(dtvr-music PUBLIC src)
target_compile_options(dtvr-music PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(dtvr-music PUBLIC dtvr-formats PkgConfig::SDL2 PkgConfig::VORBISFILE PkgConfig::JSONC Threads::Threads)
target_link_libraries(dtvr-recomb PRIVATE dtvr-music)
# Optional: text in the game window (Arial or Liberation Sans through SDL2_ttf).
pkg_check_modules(SDL2_TTF QUIET IMPORTED_TARGET SDL2_ttf)
if(SDL2_TTF_FOUND)
  target_compile_definitions(dtvr-recomb PRIVATE DTVR_HAVE_TTF)
  target_link_libraries(dtvr-recomb PRIVATE PkgConfig::SDL2_TTF)
else()
  message(STATUS "SDL2_ttf not found: the game window is built without text")
endif()
